# this yaml code and workflow is what is use to query https://svg-projects-card.onrender.com/ and create svg reponse to README.md but at the moment we cant use render raw svg to git readme file so you just have to use the single image endpoint 

name: Updating profile readme with svg-projects-card

on:
  workflow_dispatch:  # This allows manual triggering of the workflow, i see no reason updating it on cron job

jobs:
  update-readme:
    runs-on: ubuntu-latest  # Specifies the OS where the job runs
    permissions:
      contents: write

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4  # Checks out the repository so we can modify files

      - name: Fetch Project Data from API
        run: |
          # Sending a POST request with project data and saving the response to projects.json
          curl -H 'Content-Type: application/json' \
               -d '{ "repos": [
                    {
                        "name": "readme-svg-projects-card",
                        "theme": "dark"
                    },
                    {
                        "name": "Portfolio-V1",
                        "theme": "light"
                    },
                    {
                        "name": "Demi-Tasks",
                        "theme": "warm"
                    },
                    {
                        "name": "fewerCharacters",
                        "theme": "cool"
                    }
            ] }' \
            -X POST \
            https://svg-projects-card.onrender.com/api/BulusHamnu/repos \
            -o projects.json  # Saves the API response to a file

      - name: Generate Markdown with Raw SVGs
        run: |
          # Create a new projects.md file and add start tag
          echo "<!-- PROJECTS START -->" > projects.md
          
          # Extract project data from JSON and format it into markdown
          echo projects.json
          jq -r '.data[] | "<a href=\"\(.project_url)\">\(.svg)</a>"' projects.json >> projects.md
          
          # Add end tag
          echo "<!-- PROJECTS END -->" >> projects.md

      - name: Update README with projects
        run: |
          sed -i -e '/<!-- PROJECTS START -->/,/<!-- PROJECTS END -->/{
            /<!-- PROJECTS START -->/!d
            r projects.md
          }' README.md


      - name: Commit and push changes
        env:
          GH_TO: ${{ secrets.GH_TO }}  # Use the secret token
        
        run: |
          git config --global user.name "GitHub Actions"
          git config --global user.email "actions@github.com"
      
          git add README.md  # Stage the changes
      
          # Commit and push only if there are modifications
          git diff --cached --quiet || (git commit -m "Updated project SVGs" && git push https://x-access-token:${GH_TO}@github.com/BulusHamnu/BulusHamnu.git HEAD:Main)
